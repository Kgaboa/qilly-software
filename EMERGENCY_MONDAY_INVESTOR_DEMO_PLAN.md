**Add labor rate lookup to existing Qilly pricing logic:**

```typescript
// src/lib/boq/pricingEngine.ts

interface BoqPricingResult {
  materialRate: number;
  laborRate: number;      // NEW
  equipmentRate: number;  // NEW
  ohp: number;
  totalRate: number;
  totalAmount: number;
}

async function priceBoqItem(item: BoqItem): Promise<BoqPricingResult> {
  // 1. Existing material pricing (from suppliers)
  const materialRate = await getMaterialRateFromSuppliers(item.description, item.unit);
  
  // 2. NEW: Labor rate lookup
  const laborRate = await getLaborRate(item.description, item.unit);
  
  // 3. NEW: Equipment rate (for now, use 10% of composite as placeholder)
  const equipmentRate = (materialRate + laborRate) * 0.10;
  
  // 4. Calculate composite rate
  const subtotal = (materialRate || 0) + (laborRate || 0) + (equipmentRate || 0);
  const ohp = subtotal * 0.15; // 15% OH&P
  const totalRate = subtotal + ohp;
  
  return {
    materialRate: materialRate || 0,
    laborRate: laborRate || 0,
    equipmentRate: equipmentRate || 0,
    ohp,
    totalRate,
    totalAmount: totalRate * item.quantity
  };
}

async function getLaborRate(description: string, unit: string): Promise<number | null> {
  // Fuzzy match description to labor_rates_demo table
  const { data, error } = await supabase
    .from('labor_rates_demo')
    .select('labor_rate, description')
    .eq('unit', unit)
    .limit(10);
  
  if (error || !data || data.length === 0) {
    console.warn(`No labor rate found for: ${description}`);
    return null;
  }
  
  // Simple keyword matching (for demo - improve later)
  const descLower = description.toLowerCase();
  const match = data.find(rate => {
    const rateLower = rate.description.toLowerCase();
    const keywords = rateLower.split(' ');
    return keywords.some(kw => descLower.includes(kw) && kw.length > 3);
  });
  
  if (match) {
    console.log(`✅ Labor rate matched: "${description}" → "${match.description}" (R${match.labor_rate})`);
    return match.labor_rate;
  }
  
  // No match - return null (will show 0 in UI)
  return null;
}
```
