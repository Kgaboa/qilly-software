## 🎓 FEATURE GATING STRATEGY

### Implementation:
```typescript
// Feature access control
const hasFeatureAccess = (feature: string, tier: string) => {
  const tierFeatures = {
    free: ['boq_upload_training', 'basic_templates', 'pdf_export_watermark'],
    professional: [...free, 'real_pricing', 'excel_export', 'tender_response'],
    enterprise: [...professional, 'green_building', 'carbon_tracking', 'etender'],
    custom: [...enterprise, 'white_label', 'unlimited_users', 'custom_workflows']
  };
  
  return tierFeatures[tier]?.includes(feature) || false;
};
```

### Database Flag:
```sql
-- contractors table
subscription_tier: 'free' | 'professional' | 'enterprise' | 'custom'
boq_quota_used: integer  -- Track monthly usage
boq_quota_limit: integer -- 10, 30, or null (unlimited)
```
